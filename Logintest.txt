package p2;

public class Firstnonepeatativecharacter {

	public static void main(String[] args) {
		// First non repeatative character
		
		String s="vaishali";
		
		for(int i=0;i<s.length();i++)
		{
			int count=1;
			for(int j=i+1;j<s.length();j++)
			{
				if(s.charAt(i)==s.charAt(j))
				{
					count++;
				}
			}
			if(count==1)
			{
				System.out.println("First non repeatative or unique character: "+s.charAt(i));
			return;
			}
		}

	}

}
